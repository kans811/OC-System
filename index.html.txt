var PASSWORD = '1234';

function tryUnlock() {
  var input = document.getElementById('lock-input');
  var error = document.getElementById('lock-error');
  if (!input) return;

  if (input.value === PASSWORD) {
    var lock = document.getElementById('lock-screen');
    lock.classList.add('hidden');
    setTimeout(function() { lock.style.display = 'none'; }, 400);
  } else {
    error.textContent = 'Неверный код. Попробуйте ещё раз.';
    input.value = '';
    input.focus();
  }
}

var zTop = 100;
function createWindow(title, html, w, h) {
  w = w || 480; h = h || 380;
  var win = document.createElement('div');
  win.className = 'window';
  win.style.width = Math.min(w, window.innerWidth - 20) + 'px';
  win.style.height = Math.min(h, window.innerHeight - 90) + 'px';
  win.style.left = Math.max(10, (window.innerWidth - w) / 2) + 'px';
  win.style.top = Math.max(10, (window.innerHeight - h) / 2 - 30) + 'px';
  win.style.zIndex = ++zTop;
  win.innerHTML =
    '<div class="win-header"><span>' + title + '</span>' +
    '<div class="win-btns">' +
    '<button class="win-btn win-min" onclick="this.closest(\'.window\').style.display=\'none\'"></button>' +
    '<button class="win-btn win-max" onclick="maximize(this)"></button>' +
    '<button class="win-btn win-close" onclick="this.closest(\'.window\').remove()"></button>' +
    '</div></div><div class="win-body">' + html + '</div>';
  document.getElementById('desktop').appendChild(win);
  drag(win);
  win.addEventListener('mousedown', function() { win.style.zIndex = ++zTop; });
  win.addEventListener('touchstart', function() { win.style.zIndex = ++zTop; });
  return win;
}
function drag(win) {
  var head = win.querySelector('.win-header');
  var down = false, ox, oy;
  function start(x, y) { down = true; ox = x - win.offsetLeft; oy = y - win.offsetTop; }
  function move(x, y) { if (!down) return; win.style.left = (x - ox) + 'px'; win.style.top = (y - oy) + 'px'; }
  function end() { down = false; }
  head.addEventListener('mousedown', function(e) { start(e.clientX, e.clientY); });
  document.addEventListener('mousemove', function(e) { move(e.clientX, e.clientY); });
  document.addEventListener('mouseup', end);
  head.addEventListener('touchstart', function(e) { start(e.touches[0].clientX, e.touches[0].clientY); }, {passive:true});
  document.addEventListener('touchmove', function(e) { if (down) { move(e.touches[0].clientX, e.touches[0].clientY); e.preventDefault(); } }, {passive:false});
  document.addEventListener('touchend', end);
}
function maximize(btn) {
  var win = btn.closest('.window');
  if (win.dataset.max === '1') {
    win.style.width = win.dataset.ow; win.style.height = win.dataset.oh;
    win.style.left = win.dataset.ol; win.style.top = win.dataset.ot;
    win.dataset.max = '0';
  } else {
    win.dataset.ow = win.style.width; win.dataset.oh = win.style.height;
    win.dataset.ol = win.style.left; win.dataset.ot = win.style.top;
    win.style.width = '100%'; win.style.height = 'calc(100% - 55px)';
    win.style.left = '0'; win.style.top = '0';
    win.dataset.max = '1';
  }
}

function openTerminal() {
  var html = '<div class="term"><div class="term-line">WebOS Terminal v1.0</div>' +
    '<div class="term-line">Введите help для списка команд.</div>' +
    '<div id="term-out"></div>' +
    '<div class="term-line">user@webos:~$ <input type="text" id="term-in" autofocus></div></div>';
  var win = createWindow('Терминал', html, 580, 400);
  setTimeout(function() {
    var input = win.querySelector('#term-in');
    var out = win.querySelector('#term-out');
    input.focus();
    input.addEventListener('keydown', function(e) {
      if (e.key === 'Enter') {
        var cmd = input.value.trim();
        var line = document.createElement('div'); line.className = 'term-line';
        line.textContent = 'user@webos:~$ ' + cmd; out.appendChild(line);
        var res = document.createElement('div'); res.className = 'term-line';
        if (cmd === 'help') res.textContent = 'help, date, echo [текст], clear, whoami, time';
        else if (cmd === 'date') res.textContent = new Date().toLocaleString('ru-RU');
        else if (cmd === 'time') res.textContent = new Date().toLocaleTimeString('ru-RU');
        else if (cmd.indexOf('echo ') === 0) res.textContent = cmd.slice(5);
        else if (cmd === 'whoami') res.textContent = 'user';
        else if (cmd === 'clear') { out.innerHTML = ''; input.value = ''; return; }
        else if (cmd === '') res.textContent = '';
        else res.textContent = 'Команда не найдена: ' + cmd;
        out.appendChild(res); input.value = '';
        win.querySelector('.win-body').scrollTop = 9999;
      }
    });
  }, 50);
}

function brGo() {
  var input = document.getElementById('br-url');
  var q = input.value.trim();
  if (!q) return;
  var url;
  if (q.indexOf('http://') === 0 || q.indexOf('https://') === 0) { url = q; }
  else if (q.indexOf('.') !== -1 && q.indexOf(' ') === -1) { url = 'https://' + q; }
  else { url = 'https://duckduckgo.com/?q=' + encodeURIComponent(q); }
  input.value = url;
  brLoad(url);
}
function brQuick(url) {
  document.getElementById('br-url').value = url;
  brLoad(url);
}
function brLoad(url) {
  document.getElementById('br-frame').src = url;
  window.brHist = window.brHist.slice(0, window.brPos + 1);
  window.brHist.push(url);
  window.brPos = window.brHist.length - 1;
}
function brBack() {
  if (window.brPos > 0) {
    window.brPos--;
    var url = window.brHist[window.brPos];
    document.getElementById('br-frame').src = url;
    document.getElementById('br-url').value = url;
  }
}
function brForward() {
  if (window.brPos < window.brHist.length - 1) {
    window.brPos++;
    var url = window.brHist[window.brPos];
    document.getElementById('br-frame').src = url;
    document.getElementById('br-url').value = url;
  }
}
function brReload() { var f = document.getElementById('br-frame'); f.src = f.src; }
function brExt() { var url = document.getElementById('br-url').value; if (url) window.open(url, '_blank'); }

function ticMove(i) {
  if (ticOver || ticB[i] !== '') return;
  ticB[i] = 'X';
  document.getElementById('c' + i).textContent = '❌';
  if (ticWin('X')) { document.getElementById('tic-status').textContent = 'Вы победили!'; ticOver = true; return; }
  if (ticB.indexOf('') === -1) { document.getElementById('tic-status').textContent = 'Ничья'; ticOver = true; return; }
  setTimeout(function() {
    var free = [];
    for (var j = 0; j < 9; j++) if (ticB[j] === '') free.push(j);
    var m = free[Math.floor(Math.random() * free.length)];
    ticB[m] = 'O';
    document.getElementById('c' + m).textContent = '⭕';
    if (ticWin('O')) { document.getElementById('tic-status').textContent = 'Бот победил'; ticOver = true; return; }
    if (ticB.indexOf('') === -1) { document.getElementById('tic-status').textContent = 'Ничья'; ticOver = true; return; }
    document.getElementById('tic-status').textContent = 'Ваш ход (❌)';
  }, 300);
}
function ticWin(p) {
  var w = [[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];
  for (var i = 0; i < w.length; i++)
    if (ticB[w[i][0]] === p && ticB[w[i][1]] === p && ticB[w[i][2]] === p) return true;
  return false;
}

    '<button class="btn-pink" onclick="guessGo()">Проверить</button>' +
    '<div class="guess-log" id="g-log"></div></div>', 340, 380);
}
function guessGo() {
  var inp = document.getElementById('g-in');
  var log = document.getElementById('g-log');
  var v = parseInt(inp.value);
  if (isNaN(v) || v < 1 || v > 100) { log.innerHTML += '<div>Введите число от 1 до 100</div>'; return; }
  guessT++;
  if (v === guessN) { log.innerHTML += '<div>🎉 Угадали за ' + guessT + ' попыток!</div>'; inp.disabled = true; }
  else if (v < guessN) log.innerHTML += '<div>' + v + ' — больше</div>';
  else log.innerHTML += '<div>' + v + ' — меньше</div>';
  inp.value = '';
  log.scrollTop = 9999;
}

var chatHistory = [];
function openMessenger() {
  var html = '<div class="chat">' +
    '<div class="chat-head"><div class="avatar">🤖</div>' +
    '<div><b>Max Bot</b><br><span style="font-size:12px;color:#0f0;">онлайн</span></div></div>' +
    '<div class="chat-log" id="chat-log"></div>' +
    '<div class="chat-input"><input type="text" id="chat-in" placeholder="Написать сообщение...">' +
    '<button class="btn-pink" onclick="sendMsg()">➤</button></div></div>';
  createWindow('Messenger Max', html, 480, 500);

  var saved = localStorage.getItem('webos_chat');
  if (saved) chatHistory = JSON.parse(saved);
  else chatHistory = [{ t: 'Привет! Я Max. Напиши что-нибудь!', me: false }];

  var log = document.getElementById('chat-log');
  for (var i = 0; i < chatHistory.length; i++) addMsg(chatHistory[i].t, chatHistory[i].me);

  var inp = document.getElementById('chat-in');
  inp.addEventListener('keydown', function(e) {
    if (e.key === 'Enter') sendMsg();
  });
  inp.focus();
}
function addMsg(text, isMe) {
  var log = document.getElementById('chat-log');
  if (!log) return;
  var d = document.createElement('div');
  d.className = 'msg ' + (isMe ? 'user' : 'bot');
  d.textContent = text;
  log.appendChild(d);
  log.scrollTop = 9999;
}
function sendMsg() {
  var inp = document.getElementById('chat-in');
  if (!inp) return;
  var text = inp.value.trim();
  if (!text) return;
  addMsg(text, true);
  chatHistory.push({ t: text, me: true });
  saveChat();
  inp.value = '';
  var reply = botReply(text);
  addMsg(reply, false);
  chatHistory.push({ t: reply, me: false });
  saveChat();
}
function saveChat() { localStorage.setItem('webos_chat', JSON.stringify(chatHistory)); }
function botReply(text) {
  var t = text.toLowerCase();
  if (t.indexOf('привет') !== -1 || t.indexOf('здравствуй') !== -1) return 'Привет! Как дела? 😊';
  if (t.indexOf('как дела') !== -1) return 'Отлично! А у тебя?';
  if (t.indexOf('кто ты') !== -1 || t.indexOf('как тебя зовут') !== -1) return 'Я Max — бот внутри WebOS.';
  if (t.indexOf('время') !== -1 || t.indexOf('час') !== -1) return 'Сейчас ' + new Date().toLocaleTimeString('ru-RU');
  if (t.indexOf('дата') !== -1) return 'Сегодня ' + new Date().toLocaleDateString('ru-RU');
  if (t.indexOf('пока') !== -1 || t.indexOf('до свидания') !== -1) return 'Пока! Заходи ещё 👋';
  if (t.indexOf('игра') !== -1) return 'Открой «Игры» на рабочем столе!';
  if (t.indexOf('интернет') !== -1 || t.indexOf('браузер') !== -1) return 'Интернет — на рабочем столе, иконка 🌐.';
  if (t.indexOf('помощь') !== -1 || t.indexOf('help') !== -1) return 'Я умею: приветствовать, говорить время и дату, советовать игры.';
  if (t.indexOf('спасибо') !== -1) return 'Пожалуйста! 😊';
  if (t.indexOf('как') !== -1) return 'У меня всё хорошо! А у тебя?';
  if (t.indexOf('люблю') !== -1) return 'Спасибо, мне приятно! ❤️';
  if (t.indexOf('что') !== -1) return 'Хороший вопрос! Расскажи подробнее.';
  return 'Интересно! Расскажи ещё что-нибудь.';
}
