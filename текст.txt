var zTop = 100;
function createWindow(title, html, w, h) {
  w = w || 440; h = h || 340;
  var win = document.createElement('div');
  win.className = 'window';
  win.style.width = Math.min(w, window.innerWidth - 16) + 'px';
  win.style.height = Math.min(h, window.innerHeight - 80) + 'px';
  win.style.left = Math.max(8, (window.innerWidth - w) / 2) + 'px';
  win.style.top = Math.max(8, (window.innerHeight - h) / 2 - 30) + 'px';
  win.style.zIndex = ++zTop;
  win.innerHTML = '<div class="win-header"><span>' + title + '</span><div class="win-btns">' +
    '<button class="win-btn win-min" onclick="this.closest(\'.window\').style.display=\'none\'"></button>' +
    '<button class="win-btn win-max" onclick="maximize(this)"></button>' +
    '<button class="win-btn win-close" onclick="this.closest(\'.window\').remove()"></button>' +
    '</div></div><div class="win-body">' + html + '</div>';
  document.getElementById('desktop').appendChild(win);
  drag(win);
  win.addEventListener('mousedown', function() { win.style.zIndex = ++zTop; });
  win.addEventListener('touchstart', function() { win.style.zIndex = ++zTop; });
  return win;
}
function drag(win) {
  var head = win.querySelector('.win-header'), down = false, ox, oy;
  function start(x, y) { down = true; ox = x - win.offsetLeft; oy = y - win.offsetTop; }
  function move(x, y) { if (!down) return; win.style.left = (x - ox) + 'px'; win.style.top = (y - oy) + 'px'; }
  function end() { down = false; }
  head.addEventListener('mousedown', function(e) { start(e.clientX, e.clientY); });
  document.addEventListener('mousemove', function(e) { move(e.clientX, e.clientY); });
  document.addEventListener('mouseup', end);
  head.addEventListener('touchstart', function(e) { start(e.touches[0].clientX, e.touches[0].clientY); }, {passive:true});
  document.addEventListener('touchmove', function(e) { if (down) { move(e.touches[0].clientX, e.touches[0].clientY); e.preventDefault(); } }, {passive:false});
  document.addEventListener('touchend', end);
}
function maximize(btn) {
  var win = btn.closest('.window');
  if (win.dataset.max === '1') {
    win.style.width = win.dataset.ow; win.style.height = win.dataset.oh; win.style.left = win.dataset.ol; win.style.top = win.dataset.ot; win.dataset.max = '0';
  } else {
    win.dataset.ow = win.style.width; win.dataset.oh = win.style.height; win.dataset.ol = win.style.left; win.dataset.ot = win.style.top;
    win.style.width = '100%'; win.style.height = 'calc(100% - 50px)'; win.style.left = '0'; win.style.top = '0'; win.dataset.max = '1';
  }
}

function openTerminal() {
  var html = '<div class="term"><div class="term-line">WebOS Terminal</div><div class="term-line">Введите help</div><div id="term-out"></div><div class="term-line">user@webos:~$ <input id="term-in" type="text"></div></div>';
  var win = createWindow('Терминал', html, 540, 360);
  setTimeout(function() {
    var input = win.querySelector('#term-in'), out = win.querySelector('#term-out');
    input.focus();
    input.addEventListener('keydown', function(e) {
      if (e.key === 'Enter') {
        var cmd = input.value.trim();
        var line = document.createElement('div'); line.className = 'term-line'; line.textContent = 'user@webos:~$ ' + cmd; out.appendChild(line);
        var res = document.createElement('div'); res.className = 'term-line';
        if (cmd === 'help') res.textContent = 'help, date, time, echo, whoami, clear';
        else if (cmd === 'date') res.textContent = new Date().toLocaleString('ru-RU');
        else if (cmd === 'time') res.textContent = new Date().toLocaleTimeString('ru-RU');
        else if (cmd.indexOf('echo ') === 0) res.textContent = cmd.slice(5);
        else if (cmd === 'whoami') res.textContent = 'user';
        else if (cmd === 'clear') { out.innerHTML = ''; input.value = ''; return; }
        else if (cmd !== '') res.textContent = 'Не найдено: ' + cmd;
        out.appendChild(res); input.value = '';
      }
    });
  }, 50);
}

function addMsg(text, isMe) {
  var log = document.getElementById('chat-log'); if (!log) return;
  var d = document.createElement('div'); d.className = 'msg ' + (isMe ? 'user' : 'bot'); d.textContent = text;
  log.appendChild(d); log.scrollTop = 9999;
}
function sendMsg() {
  var inp = document.getElementById('chat-in'); if (!inp) return;
  var text = inp.value.trim(); if (!text) return;
  addMsg(text, true);
  var saved = localStorage.getItem('webos_chat');
  var hist = saved ? JSON.parse(saved) : [{t:'Привет! Я Max. Напиши что-нибудь!',me:false}];
  hist.push({t:text,me:true});
  inp.value = '';
  var reply = botReply(text);
  addMsg(reply, false);
  hist.push({t:reply,me:false});
  localStorage.setItem('webos_chat', JSON.stringify(hist));
}
function botReply(text) {
  var t = text.toLowerCase();
  var math = text.match(/^[\d\s\+\-\*\/\.\(\)]+$/);
  if (math) { try { return 'Ответ: ' + eval(text); } catch(e) {} }
  if (t.indexOf('привет') !== -1) return 'Привет! Как дела? 😊';
  if (t.indexOf('как дела') !== -1) return 'Отлично! А у тебя?';
  if (t.indexOf('кто ты') !== -1) return 'Я Max — бот внутри WebOS.';
  if (t.indexOf('время') !== -1) return 'Сейчас ' + new Date().toLocaleTimeString('ru-RU');
  if (t.indexOf('дата') !== -1) return 'Сегодня ' + new Date().toLocaleDateString('ru-RU');
  if (t.indexOf('пока') !== -1) return 'Пока! Заходи ещё 👋';
  if (t.indexOf('спасибо') !== -1) return 'Пожалуйста! 😊';
  return 'Интересно! Расскажи подробнее.';
}

}

function openCalc() {
  var keys = ['7','8','9','/','4','5','6','*','1','2','3','-','0','.','=','+'];
  var html = '<div class="calc"><div class="calc-display" id="calc-disp">0</div>' +
    '<button class="calc-btn" onclick="calcIn(\'C\')" style="grid-column:span 4;">C</button>';
  keys.forEach(function(k) { html += '<button class="calc-btn' + (k==='='?' eq':'') + '" onclick="calcIn(\'' + k + '\')">' + k + '</button>'; });
  html += '</div>';
  createWindow('Калькулятор', html, 300, 400);
  window.calcBuf = '0';
}
function calcIn(v) {
  var disp = document.getElementById('calc-disp'); if (!disp) return;
  if (v === 'C') window.calcBuf = '0';
  else if (v === '=') { try { window.calcBuf = String(eval(window.calcBuf) || 0); } catch(e) { window.calcBuf = 'Ошибка'; } }
  else { window.calcBuf = (window.calcBuf === '0' && v !== '.') ? v : window.calcBuf + v; }
  disp.textContent = window.calcBuf;
}

var calMonth, calYear;
function openCalendar() {
  var d = new Date(); calMonth = d.getMonth(); calYear = d.getFullYear();
  var html = '<div class="cal-head"><button class="btn-gray" onclick="calShift(-1)">◄</button><span id="cal-title" style="font-weight:bold;"></span><button class="btn-gray" onclick="calShift(1)">►</button></div><div class="cal-grid" id="cal-grid"></div>';
  createWindow('Календарь', html, 320, 360);
  calRender();
}
function calShift(d) { calMonth += d; if (calMonth < 0) { calMonth = 11; calYear--; } if (calMonth > 11) { calMonth = 0; calYear++; } calRender(); }
function calRender() {
  var months = ['Январь','Февраль','Март','Апрель','Май','Июнь','Июль','Август','Сентябрь','Октябрь','Ноябрь','Декабрь'];
  document.getElementById('cal-title').textContent = months[calMonth] + ' ' + calYear;
  var html = '', days = ['Пн','Вт','Ср','Чт','Пт','Сб','Вс'];
  for (var i = 0; i < 7; i++) html += '<div class="cal-day-name">' + days[i] + '</div>';
  var first = new Date(calYear, calMonth, 1).getDay(); if (first === 0) first = 7;
  var total = new Date(calYear, calMonth + 1, 0).getDate();
  for (var i = 1; i < first; i++) html += '<div></div>';
  var today = new Date();
  for (var day = 1; day <= total; day++) {
    var isT = (day === today.getDate() && calMonth === today.getMonth() && calYear === today.getFullYear());
    html += '<div class="cal-day' + (isT ? ' today' : '') + '">' + day + '</div>';
  }
  document.getElementById('cal-grid').innerHTML = html;
}

var timerInt, timerSec = 0, timerRun = false;
function openTimer() {
  createWindow('Таймер', '<div class="timer-display" id="timer-disp">00:00</div><div class="timer-btns"><button class="btn-green" id="timer-btn" onclick="timerToggle()">Старт</button><button class="btn-gray" onclick="timerReset()">Сброс</button></div>', 280, 220);
  timerUpd();
}
function timerToggle() {
  if (timerRun) { clearInterval(timerInt); timerRun = false; document.getElementById('timer-btn').textContent = 'Старт'; document.getElementById('timer-btn').className = 'btn-green'; }
  else { timerRun = true; document.getElementById('timer-btn').textContent = 'Пауза'; document.getElementById('timer-btn').className = 'btn-gray'; timerInt = setInterval(function() { timerSec++; timerUpd(); }, 1000); }
}
function timerReset() { clearInterval(timerInt); timerRun = false; timerSec = 0; timerUpd(); var b = document.getElementById('timer-btn'); if (b) { b.textContent = 'Старт'; b.className = 'btn-green'; } }
function timerUpd() { var d = document.getElementById('timer-disp'); if (d) d.textContent = ('0' + Math.floor(timerSec/60)).slice(-2) + ':' + ('0' + timerSec%60).slice(-2); }

var shopList = JSON.parse(localStorage.getItem('webos_shop') || '[]');
function openShop() {
  createWindow('Покупки', '<div class="shop-input"><input id="shop-in" type="text" placeholder="Что купить?"><button class="btn-pink" onclick="shopAdd()">+</button></div><div id="shop-list"></div>', 360, 400);
  shopRender();
  setTimeout(function() { var i = document.getElementById('shop-in'); if (i) i.addEventListener('keydown', function(e) { if (e.key === 'Enter') shopAdd(); }); }, 50);
}
function shopAdd() { var inp = document.getElementById('shop-in'); if (!inp || !inp.value.trim()) return; shopList.push({text:inp.value.trim(),done:false}); inp.value=''; shopSave(); shopRender(); }
function shopRender() {
  var el = document.getElementById('shop-list'); if (!el) return;
  if (shopList.length === 0) { el.innerHTML = '<div style="color:#888;text-align:center;padding:20px;">Список пуст</div>'; return; }
  el.innerHTML = '';
  shopList.forEach(function(item, i) {
    el.innerHTML += '<div class="shop-item' + (item.done ? ' done' : '') + '"><span onclick="shopToggle(' + i + ')" style="flex:1;cursor:pointer;">' + (item.done ? '✓ ' : '') + item.text + '</span><button class="shop-del" onclick="shopDel(' + i + ')">✕</button></div>';
  });
}
function shopToggle(i) { shopList[i].done = !shopList[i].done; shopSave(); shopRender(); }
function shopDel(i) { shopList.splice(i, 1); shopSave(); shopRender(); }
function shopSave() { localStorage.setItem('webos_shop', JSON.stringify(shopList)); }

function openSettings() {
  createWindow('Настройки', '<div class="set-row"><label>Обои:</label><button class="btn-gray" onclick="setWall(1)">Синие</button> <button class="btn-gray" onclick="setWall(2)">Фиолетовые</button> <button class="btn-gray" onclick="setWall(3)">Зелёные</button></div><div class="set-row"><label>Сброс:</label><button class="btn-gray" onclick="resetAll()">Очистить всё</button></div>', 360, 260);
}
function setWall(n) {
  var css = n===1 ? 'linear-gradient(135deg,#0f3460,#16213e,#533483)' : n===2 ? 'linear-gradient(135deg,#2b1055,#7597de)' : 'linear-gradient(135deg,#134e5e,#71b280)';
  document.getElementById('desktop').style.background = css;
  localStorage.setItem('webos_wall', css);
}
function resetAll() { if (confirm('Удалить все данные?')) { localStorage.clear(); location.reload(); } }
